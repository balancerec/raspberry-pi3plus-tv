cmake_minimum_required(VERSION 3.16)
project(raspberry_tv_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(tv_input STATIC src/input.cpp)
target_include_directories(tv_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(tv_input PRIVATE -Wall -Wextra -Wpedantic)

option(BUILD_NATIVE_APP "Build the SDL2 controller-driven prototype" ON)
if(BUILD_NATIVE_APP)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    add_executable(raspberry-tv-native src/main.cpp)
    target_link_libraries(raspberry-tv-native PRIVATE tv_input PkgConfig::SDL2)
    target_compile_options(raspberry-tv-native PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tv-input-tests tests/input_tests.cpp)
    target_link_libraries(tv-input-tests PRIVATE tv_input)
    add_test(NAME controller_input COMMAND tv-input-tests)
endif()
