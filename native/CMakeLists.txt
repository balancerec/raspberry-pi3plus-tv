cmake_minimum_required(VERSION 3.16)
project(raspberry_tv_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(tv_input STATIC src/input.cpp)
target_include_directories(tv_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(tv_input PRIVATE -Wall -Wextra -Wpedantic)

add_library(tv_menu STATIC src/menu.cpp)
target_include_directories(tv_menu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(tv_menu PUBLIC tv_input)
target_compile_options(tv_menu PRIVATE -Wall -Wextra -Wpedantic)

option(BUILD_NATIVE_APP "Build the SDL2 gamepad-first television interface" ON)
if(BUILD_NATIVE_APP)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
    add_executable(raspberry-tv-native src/main.cpp src/ui.cpp)
    target_link_libraries(raspberry-tv-native PRIVATE tv_input tv_menu PkgConfig::SDL2 PkgConfig::SDL2_TTF)
    target_compile_options(raspberry-tv-native PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tv-input-tests tests/input_tests.cpp)
    target_link_libraries(tv-input-tests PRIVATE tv_input)
    add_test(NAME controller_input COMMAND tv-input-tests)
    add_executable(tv-menu-tests tests/menu_tests.cpp)
    target_link_libraries(tv-menu-tests PRIVATE tv_menu)
    add_test(NAME menu_navigation_and_m3u COMMAND tv-menu-tests)
    add_executable(tv-display-tests tests/display_tests.cpp)
    target_include_directories(tv-display-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_test(NAME fullhd_native_pixels COMMAND tv-display-tests)
endif()
