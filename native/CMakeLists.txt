cmake_minimum_required(VERSION 3.16)
project(raspberry_tv_native VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
function(tv_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
add_library(tv_input STATIC src/input.cpp)
target_include_directories(tv_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
tv_warnings(tv_input)
add_library(tv_menu STATIC src/menu.cpp src/playlist.cpp src/playlist_loader.cpp src/settings.cpp src/state_writer.cpp src/playback_state.cpp src/spotify.cpp)
target_include_directories(tv_menu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(tv_menu PUBLIC tv_input Threads::Threads)
tv_warnings(tv_menu)
option(BUILD_NATIVE_APP "Build the SDL2/X11 GStreamer television application" ON)
if(BUILD_NATIVE_APP)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
    pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.20)
    pkg_check_modules(GSTREAMER_VIDEO REQUIRED IMPORTED_TARGET gstreamer-video-1.0)
    add_library(tv_player STATIC src/player.cpp)
    target_link_libraries(tv_player PUBLIC tv_menu PkgConfig::SDL2 PkgConfig::GSTREAMER PkgConfig::GSTREAMER_VIDEO)
    tv_warnings(tv_player)
    add_executable(raspberry-tv-native src/main.cpp src/ui.cpp src/spotify_process.cpp)
    target_link_libraries(raspberry-tv-native PRIVATE tv_player PkgConfig::SDL2_TTF)
    tv_warnings(raspberry-tv-native)
    install(TARGETS raspberry-tv-native RUNTIME DESTINATION bin)
    install(FILES ../config/defaults.conf DESTINATION share/raspberry-tv)
    install(FILES ../integrations/spotify/spotify_client.py DESTINATION integrations/spotify)
    install(FILES ../scripts/diagnose-spotify.py DESTINATION scripts)
    install(FILES ../scripts/desktop-launcher.py DESTINATION scripts)
endif()

if(BUILD_TESTING)
    function(tv_test target source name)
        add_executable(${target} ${source})
        target_link_libraries(${target} PRIVATE tv_menu)
        tv_warnings(${target})
        add_test(NAME ${name} COMMAND ${target})
        set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endfunction()
    tv_test(tv-input-tests tests/input_tests.cpp controller_input)

    tv_test(tv-persistence-tests tests/persistence_tests.cpp atomic_settings_and_favorites)
    tv_test(tv-playback-state-tests tests/playback_state_tests.cpp playback_timeouts_retries_and_cancel)
